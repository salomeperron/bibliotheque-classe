<script>

const STORAGE_KEY="bibliotheque_classe_v1";

function defaultData(){

    return {

        livres:[],

        eleves:[],

        emprunts:[],

        commentaires:[]

    };

}

function getData(){

    return JSON.parse(

        localStorage.getItem(STORAGE_KEY)

        || JSON.stringify(defaultData())

    );

}

function saveData(data){

    localStorage.setItem(

        STORAGE_KEY,

        JSON.stringify(data)

    );

}

function connexion(){

    const nom=

        document.getElementById("userName").value.trim();

    if(!nom) return;

    localStorage.setItem("user_actuel",nom);

    afficherUtilisateur();

}

function utilisateur(){

    return localStorage.getItem("user_actuel") || "";

}

function afficherUtilisateur(){

    document.getElementById("currentUser")

        .innerHTML=

        utilisateur()

        ? "Connecté : <strong>"+utilisateur()+"</strong>"

        : "Aucun utilisateur";

}

function ajouterLivre(){

    const titre=

    document.getElementById("livreTitre").value.trim();

    const auteur=

    document.getElementById("livreAuteur").value.trim();

    if(!titre) return;

    const data=getData();

    data.livres.push({

        id:Date.now(),

        titre,

        auteur,

        disponible:true

    });

    saveData(data);

    document.getElementById("livreTitre").value="";

    document.getElementById("livreAuteur").value="";

    render();

}

function ajouterEleve(){

    const nom=

    document.getElementById("eleveNom").value.trim();

    if(!nom) return;

    const data=getData();

    data.eleves.push({

        id:Date.now(),

        nom

    });

    saveData(data);

    document.getElementById("eleveNom").value="";

    render();

}

function emprunter(id){

    const user=utilisateur();

    if(!user){

        alert("Connectez-vous d'abord.");

        return;

    }

    const data=getData();

    const livre=data.livres.find(l=>l.id===id);

    livre.disponible=false;

    data.emprunts.push({

        livreId:id,

        utilisateur:user,

        date:new Date().toLocaleDateString()

    });

    saveData(data);

    render();

}

function retourner(id){

    const data=getData();

    const livre=data.livres.find(l=>l.id===id);

    livre.disponible=true;

    data.emprunts=

        data.emprunts.filter(

            e=>e.livreId!==id

        );

    saveData(data);

    render();

}

function ajouterCommentaire(){

    const livreId=

        Number(

            document.getElementById(

            "commentLivre"

            ).value

        );

    const texte=

        document.getElementById(

            "commentTexte"

        ).value.trim();

    if(!texte) return;

    const data=getData();

    data.commentaires.push({

        livreId,

        auteur:utilisateur() || "Anonyme",

        texte

    });

    saveData(data);

    document.getElementById(

        "commentTexte"

    ).value="";

    render();

}

function exporter(){

    const data=

        localStorage.getItem(STORAGE_KEY);

    const blob=

        new Blob([data],{

            type:"application/json"

        });

    const a=

        document.createElement("a");

    a.href=

        URL.createObjectURL(blob);

    a.download=

        "bibliotheque-classe.json";

    a.click();

}

function importer(){

    const file=

        document.getElementById(

            "importFile"

        ).files[0];

    if(!file) return;

    const reader=

        new FileReader();

    reader.onload=e=>{

        localStorage.setItem(

            STORAGE_KEY,

            e.target.result

        );

        render();

    };

    reader.readAsText(file);

}

function render(){

    afficherUtilisateur();

    const data=getData();

    const tbody=

    document.getElementById("livresTable");

    tbody.innerHTML="";

    data.livres.forEach(l=>{

        const tr=document.createElement("tr");

        tr.innerHTML=`

        <td>${l.titre}</td>

        <td>${l.auteur||''}</td>

        <td>

            <span class="badge">

            ${l.disponible?'Disponible':'Emprunté'}

            </span>

        </td>

        <td>

        ${

            l.disponible

            ?

            `<button class="primary"

            onclick="emprunter(${l.id})">

            Emprunter

            </button>`

            :

            `<button class="danger"

            onclick="retourner(${l.id})">

            Retour

            </button>`

        }

        </td>

        `;

        tbody.appendChild(tr);

    });

    document.getElementById(

        "elevesListe"

    ).innerHTML=

        data.eleves

        .map(e=>`<li>${e.nom}</li>`)

        .join("");

    const select=

    document.getElementById("commentLivre");

    select.innerHTML=

        data.livres

        .map(l=>`

        <option value="${l.id}">

        ${l.titre}

        </option>

        `).join("");

    document.getElementById(

        "commentaires"

    ).innerHTML=

        data.commentaires.map(c=>{

            const livre=

            data.livres.find(

                l=>l.id===c.livreId

            );

            return `

            <div class="card">

                <strong>

                ${livre?livre.titre:'Livre'}

                </strong><br>

                ${c.auteur}<br>

                ${c.texte}

            </div>

            `;

        }).join("");

}

render();
